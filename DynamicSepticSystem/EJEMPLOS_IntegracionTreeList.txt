// EJEMPLO 8: VERIFICAR DISPONIBILIDAD
// ======================================

private bool EditorTreeListDisponible()
{
    try
    {
        string conn = @"Server=.\SQLEXPRESS;Database=BaseDatosCalandria;Trusted_Connection=True;";
        
        using (var sqlConn = new System.Data.SqlClient.SqlConnection(conn))
        {
            sqlConn.Open();
            
            var cmd = new System.Data.SqlClient.SqlCommand(
                "SELECT COUNT(*) FROM sys.objects WHERE name = 'TreeListData'", sqlConn);
            
            int count = (int)cmd.ExecuteScalar();
            return count > 0;
        }
    }
    catch
    {
        return false;
    }
}

// EJEMPLO 9: OBTENER ESTADÍSTICAS
// ==================================

private string ObtenerEstadisticasTreeList()
{
    try
    {
        string conn = @"Server=.\SQLEXPRESS;Database=BaseDatosCalandria;Trusted_Connection=True;";
        
        using (var sqlConn = new System.Data.SqlClient.SqlConnection(conn))
        {
            sqlConn.Open();
            
            var cmd = new System.Data.SqlClient.SqlCommand(@"
                SELECT 
                    COUNT(*) as Total,
                    SUM(CASE WHEN Nivel = 0 THEN 1 ELSE 0 END) as Padres,
                    SUM(CASE WHEN Nivel = 1 THEN 1 ELSE 0 END) as SubPadres,
                    SUM(CASE WHEN Nivel = 2 THEN 1 ELSE 0 END) as Hijos
                FROM TreeListData", sqlConn);
            
            using (var reader = cmd.ExecuteReader())
            {
                if (reader.Read())
                {
                    return $"Total: {reader["Total"]}\n" +
                           $"Padres: {reader["Padres"]}\n" +
                           $"Sub-Padres: {reader["SubPadres"]}\n" +
                           $"Hijos: {reader["Hijos"]}";
                }
            }
        }
    }
    catch (Exception ex)
    {
        return $"Error: {ex.Message}";
    }
    
    return "Sin datos";
}

// EJEMPLO 10: CONFIGURACIÓN AVANZADA
// =====================================

private void AbrirEditorConfigurado()
{
    string conn = @"Server=.\SQLEXPRESS;Database=BaseDatosCalandria;Trusted_Connection=True;";
    
    var form = new FormEditorTreeList(conn)
    {
        Text = "Gestor de Proyectos y Tareas",
        StartPosition = FormStartPosition.CenterScreen,
        WindowState = FormWindowState.Maximized
    };
    
    // Evento al cargar
    form.Load += (s, e) => 
    {
        MessageBox.Show("¡Bienvenido al Editor de Tareas!", 
            "Información", MessageBoxButtons.OK, MessageBoxIcon.Information);
    };
    
    // Evento al cerrar
    form.FormClosing += (s, e) =>
    {
        var result = MessageBox.Show("¿Seguro que deseas salir?", 
            "Confirmar", MessageBoxButtons.YesNo, MessageBoxIcon.Question);
        
        if (result != DialogResult.Yes)
        {
            e.Cancel = true;
        }
    };
    
    form.ShowDialog();
}
